cmake_minimum_required(VERSION 3.22)
project(lami_tokenizer_only LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
if(NOT SENTENCEPIECE_SOURCE)
  message(FATAL_ERROR "Use scripts/build_tokenizer_only.py with a verified source checkout")
endif()
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${SENTENCEPIECE_SOURCE}" sentencepiece EXCLUDE_FROM_ALL)
add_library(lami_tokenizer_only SHARED tokenizer_jni.cc)
target_include_directories(lami_tokenizer_only PRIVATE "${SENTENCEPIECE_SOURCE}/src")
if(NOT ANDROID)
  find_package(JNI REQUIRED)
  target_include_directories(lami_tokenizer_only PRIVATE ${JNI_INCLUDE_DIRS})
endif()
target_link_libraries(lami_tokenizer_only PRIVATE sentencepiece-static)
if(ANDROID)
  target_link_libraries(lami_tokenizer_only PRIVATE log)
endif()
target_link_options(lami_tokenizer_only PRIVATE
  "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map"
  "-Wl,--exclude-libs,ALL" "-Wl,-Bsymbolic" "-Wl,-z,max-page-size=16384")
set_property(TARGET lami_tokenizer_only APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
